======================================================================
MATLAB FILE FORMAT
======================================================================

The generation script is:

--------------------------------------------------

%% H

for row = 1 : Nr
    for col = 1 : Nt
        fprintf(fid_H,'%s\n',bin(real(H(row,col))));
        fprintf(fid_H,'%s\n',bin(imag(H(row,col))));
    end
end

%% Y

for i = 1 : Nr
    fprintf(fid_Y,'%s\n',bin(real(Y(i))));
    fprintf(fid_Y,'%s\n',bin(imag(Y(i))));
end

%% Z

for i = 1 : Nr
    fprintf(fid_Z,'%s\n',bin(real(Z(i))));
    fprintf(fid_Z,'%s\n',bin(imag(Z(i))));
end

--------------------------------------------------

Interpretation:

======================================================================
H FILE
======================================================================

Frame format:

    H(0,0).real
    H(0,0).imag

    H(0,1).real
    H(0,1).imag

    H(7,7).real
    H(7,7).imag

Row-major ordering.

For 8x8:

    64 complex entries

    Q1.11
